cmake_minimum_required(VERSION 3.16)
project(cotocoa_install_consumer LANGUAGES C)
option(ENABLE_FORTRAN "Exercise the installed Fortran component" OFF)
if(ENABLE_FORTRAN)
    enable_language(Fortran)
    find_package(cotocoa 0.3 REQUIRED COMPONENTS Fortran)
    add_executable(consumer_fortran fortran.f90)
    target_link_libraries(consumer_fortran PRIVATE cotocoa::cotocoa_fortran)
else()
    find_package(cotocoa 0.3 REQUIRED)
endif()
add_executable(consumer_c c.c)
target_link_libraries(consumer_c PRIVATE cotocoa::cotocoa)
